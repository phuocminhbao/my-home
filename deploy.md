# How to deploy

## Prerequire

-   A VPC install git, dopcker and docker compose

### 1. Set up folders/files structure in your vpc

    ├──
    |
    └──

    /vpc-root (e.g: AWS EC2 is /home/ec2-user)
    │
    ├──/certs (This folder will be automactic generated when running docker compose)
    │ └──...
    │
    ├── /nginx # Nginx folder containing configuration
    │ ├── nginx.conf # Nginx configuration file
    | └── mime.types
    │
    ├── docker-compose.yml # Docker Compose file for handling containers
    └── ...

### 2. Set up docker compose

docker-compose.yml:
version: "3.3"

    services:
        app:
        image: 31200/my-home:1 (Replace with newest image of the app)
        container_name: vantridhc
        restart: unless-stopped
        expose: - "3000" # Expose app port to Nginx
        networks: - webnet

        nginx:
            image: nginx:latest
            container_name: nginx
            restart: always
            ports:
                - "80:80" # HTTP
                - "443:443" # HTTPS
            depends_on:
                - app
            volumes:
                - ./nginx:/etc/nginx/:ro # Mount your Nginx configuration
                - ./certs:/etc/letsencrypt/:ro # Mount for certificates
                - ./certs/www:/var/www/_letsencrypt/:ro
            networks:
                - webnet

        certbot:
            image: certbot/certbot:latest
            container_name: certbot
            volumes:
                - ./nginx:/etc/nginx/:rw # Mount your Nginx configuration
                - ./certs:/etc/letsencrypt/:rw # Mount for certificates
                - ./certs/www:/var/www/_letsencrypt/:rw
            entrypoint: |
                /bin/sh -c 'trap exit TERM; while :; do certbot renew ; sleep 12h & wait $${!}; done;'
            networks:
                - webnet

    networks:
        webnet:
            driver: bridge

### 3. Set up nginx config

Could using the nginx tool like digitalocean or
Edit the /nginx/nginx.conf (create if not existed):

    # Generated by nginxconfig.io
    # See nginxconfig.txt for the configuration share link

    # user                 vantri;
    pid                  /run/nginx.pid;
    worker_processes     auto;
    worker_rlimit_nofile 65535;

    # Load modules
    include              /etc/nginx/modules-enabled/*.conf;

    events {
        multi_accept       on;
        worker_connections 65535;
    }

    http {
        charset                utf-8;
        sendfile               on;
        tcp_nopush             on;
        tcp_nodelay            on;
        server_tokens          off;
        log_not_found          off;
        types_hash_max_size    2048;
        types_hash_bucket_size 64;
        client_max_body_size   16M;

        # MIME
        include                mime.types;
        default_type           application/octet-stream;

        # Logging
        access_log             off;
        error_log              /dev/null;

        # SSL
        ssl_session_timeout    1d;
        ssl_session_cache      shared:SSL:10m;
        ssl_session_tickets    off;

        # Diffie-Hellman parameter for DHE ciphersuites
        ssl_dhparam            /etc/nginx/dhparam.pem;

        # Mozilla Intermediate configuration
        ssl_protocols          TLSv1.2 TLSv1.3;
        ssl_ciphers            ECDHE-ECDSA-AES128-GCM-SHA256:ECDHE-RSA-AES128-GCM-SHA256:ECDHE-ECDSA-AES256-GCM-SHA384:ECDHE-RSA-AES256-GCM-SHA384:ECDHE-ECDSA-CHACHA20-POLY1305:ECDHE-RSA-CHACHA20-POLY1305:DHE-RSA-AES128-GCM-SHA256:DHE-RSA-AES256-GCM-SHA384;

        # OCSP Stapling
        ssl_stapling           on;
        ssl_stapling_verify    on;
        resolver               1.1.1.1 1.0.0.1 8.8.8.8 8.8.4.4 208.67.222.222 208.67.220.220 valid=60s;
        resolver_timeout       2s;

        # Connection header for WebSocket reverse proxy
        map $http_upgrade $connection_upgrade {
            default upgrade;
            ""      close;
        }

        map $remote_addr $proxy_forwarded_elem {

            # IPv4 addresses can be sent as-is
            ~^[0-9.]+$        "for=$remote_addr";

            # IPv6 addresses need to be bracketed and quoted
            ~^[0-9A-Fa-f:.]+$ "for=\"[$remote_addr]\"";

            # Unix domain socket names cannot be represented in RFC 7239 syntax
            default           "for=unknown";
        }

        map $http_forwarded $proxy_add_forwarded {

            # If the incoming Forwarded header is syntactically valid, append to it
            "~^(,[ \\t]*)*([!#$%&'*+.^_`|~0-9A-Za-z-]+=([!#$%&'*+.^_`|~0-9A-Za-z-]+|\"([\\t \\x21\\x23-\\x5B\\x5D-\\x7E\\x80-\\xFF]|\\\\[\\t \\x21-\\x7E\\x80-\\xFF])*\"))?(;([!#$%&'*+.^_`|~0-9A-Za-z-]+=([!#$%&'*+.^_`|~0-9A-Za-z-]+|\"([\\t \\x21\\x23-\\x5B\\x5D-\\x7E\\x80-\\xFF]|\\\\[\\t \\x21-\\x7E\\x80-\\xFF])*\"))?)*([ \\t]*,([ \\t]*([!#$%&'*+.^_`|~0-9A-Za-z-]+=([!#$%&'*+.^_`|~0-9A-Za-z-]+|\"([\\t \\x21\\x23-\\x5B\\x5D-\\x7E\\x80-\\xFF]|\\\\[\\t \\x21-\\x7E\\x80-\\xFF])*\"))?(;([!#$%&'*+.^_`|~0-9A-Za-z-]+=([!#$%&'*+.^_`|~0-9A-Za-z-]+|\"([\\t \\x21\\x23-\\x5B\\x5D-\\x7E\\x80-\\xFF]|\\\\[\\t \\x21-\\x7E\\x80-\\xFF])*\"))?)*)?)*$" "$http_forwarded, $proxy_forwarded_elem";

            # Otherwise, replace it
            default "$proxy_forwarded_elem";
        }

        # Load configs
        include /etc/nginx/conf.d/*.conf;

        # vantridhc.com
        server {
            listen                               443 ssl;
            listen                               [::]:443 ssl;
            server_name                          vantridhc.com;

            # SSL
            ssl_certificate                      /etc/letsencrypt/live/vantridhc.com/fullchain.pem;
            ssl_certificate_key                  /etc/letsencrypt/live/vantridhc.com/privkey.pem;
            ssl_trusted_certificate              /etc/letsencrypt/live/vantridhc.com/chain.pem;

            # security headers
            add_header X-XSS-Protection          "1; mode=block" always;
            add_header X-Content-Type-Options    "nosniff" always;
            add_header Referrer-Policy           "no-referrer-when-downgrade" always;
            add_header Content-Security-Policy   "default-src 'self' http: https: ws: wss: data: blob: 'unsafe-inline'; frame-ancestors 'self';" always;
            add_header Permissions-Policy        "interest-cohort=()" always;
            add_header Strict-Transport-Security "max-age=31536000; includeSubDomains" always;

            # . files
            location ~ /\.(?!well-known) {
                deny all;
            }

            # reverse proxy
            location / {
                proxy_pass                         http://app:3000;
                proxy_set_header Host              $host;
                proxy_http_version                 1.1;
                proxy_cache_bypass                 $http_upgrade;

                # Proxy SSL
                proxy_ssl_server_name              on;

                # Proxy headers
                proxy_set_header Upgrade           $http_upgrade;
                proxy_set_header Connection        $connection_upgrade;
                proxy_set_header X-Real-IP         $remote_addr;
                proxy_set_header Forwarded         $proxy_add_forwarded;
                proxy_set_header X-Forwarded-For   $proxy_add_x_forwarded_for;
                proxy_set_header X-Forwarded-Proto $scheme;
                proxy_set_header X-Forwarded-Host  $host;
                proxy_set_header X-Forwarded-Port  $server_port;

                # Proxy timeouts
                proxy_connect_timeout              60s;
                proxy_send_timeout                 60s;
                proxy_read_timeout                 60s;
            }

            # favicon.ico
            location = /favicon.ico {
                log_not_found off;
            }

            # robots.txt
            location = /robots.txt {
                log_not_found off;
            }

            # gzip
            gzip            on;
            gzip_vary       on;
            gzip_proxied    any;
            gzip_comp_level 6;
            gzip_types      text/plain text/css text/xml application/json application/javascript application/rss+xml application/atom+xml image/svg+xml;
        }

        # subdomains redirect
        server {
            listen                  443 ssl;
            listen                  [::]:443 ssl;
            server_name             *.vantridhc.com;

            # SSL
            ssl_certificate         /etc/letsencrypt/live/vantridhc.com/fullchain.pem;
            ssl_certificate_key     /etc/letsencrypt/live/vantridhc.com/privkey.pem;
            ssl_trusted_certificate /etc/letsencrypt/live/vantridhc.com/chain.pem;
            return                  301 https://vantridhc.com$request_uri;
        }

        # HTTP redirect
        server {
            listen      80;
            listen      [::]:80;
            server_name .vantridhc.com;

            # ACME-challenge
            location ^~ /.well-known/acme-challenge/ {
                root /var/www/_letsencrypt;
            }

            location / {
                return 301 https://vantridhc.com$request_uri;
            }
        }
    }

### 4. Init SSL with cerbot for the first time (Skip if SSL inited)

1. Edit the nginx/nginx.conf => Comments out or cut the 2 server 443 block inside http block (This require because nginx need to be running and allow port 80 for certbot init ssl)
2. CD to root where docker-compose located, run docker-compose up
   Verify all services running (docker ps -a)
3. Run command: docker-compse exec <certbot's container id> sh
   run: "certbot certonly --webroot -d vantridhc.com --email huuphuoc030100@gmail.com -w /var/www/\_letsencrypt -n --agree-tos --force-renewal"
4. If it run success (There will message: Successfully received certificate.)
   SSL files will generated in /certs folder
5. Edit the nginx/nginx.conf => Bring back the 2 server 443 block inside http block => save

Everything should working now, if not debug at a dev :)
